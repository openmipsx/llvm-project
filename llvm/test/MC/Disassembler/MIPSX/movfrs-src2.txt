# movfrs keeps Dest in the Src2 field, bits 21-17; bits 16-12 are 0
# (processor spec §2, §6, "movfrs destination"; devices b6cdbcb).
# RUN: llvm-mc -triple=mipsx -disassemble %s 2>/dev/null | FileCheck %s
# RUN: llvm-mc -triple=mipsx -disassemble %s 2>&1 >/dev/null \
# RUN:   | FileCheck --check-prefix=ERR %s

# OP 011 compute-immediate, Spec 1 (psw), r2 in bits 21-17.
0xd8 0x04 0x00 0x01
# CHECK: movfrs psw,r2
# CHECK-NOT: {{[a-z]}}

# The same with r2 in bits 16-12 instead, as earlier versions of the spec
# had it: not an instruction.
0xd8 0x00 0x20 0x01
